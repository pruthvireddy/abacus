def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as file:
        for block in iter(lambda:file.read(1024*1024),b''):h.update(block)
    return h.hexdigest()
def pin(relative):
    path=stage/relative
    if Path(relative).is_absolute() or '..' in Path(relative).parts or path.is_symlink() or not path.is_file():raise ValueError('ordinary stage file required')
    return dict(path=relative,bytes=path.stat().st_size,sha256=sha(path))
def write(relative,value):
    path=stage/relative;path.parent.mkdir(parents=True,exist_ok=True)
    with path.open('x') as file:json.dump(value,file,indent=2,allow_nan=False);file.write('\n')
def copied(source,relative):
    target=stage/relative;target.parent.mkdir(parents=True,exist_ok=True)
    if target.exists():raise FileExistsError('new evidence copy only')
    shutil.copyfile(source,target)
    if sha(source)!=sha(target):raise ValueError('copied evidence mismatch')
    return pin(relative)
